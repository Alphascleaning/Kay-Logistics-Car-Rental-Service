Customer Information // ----------------------------- doc.setFontSize(11); doc.text("Customer Information", 10, 40); doc.line(10, 42, 200, 42); doc.text(`Full Name: ${name}`, 10, 50); doc.text(`Phone Number: ${phone}`, 10, 58); doc.text(`Email: ${email}`, 10, 66); doc.text(`DI#/Passport#: ${idNumber}`, 10, 74); // ----------------------------- // 5.
